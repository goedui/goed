Goed UI 组件库

纯 Go 实现的桌面 UI 组件体系，为编辑器与工具类应用而生

所有组件为保留模式绘制，接收 core.DrawingContext 抽象渲染，主题经 ui/theme 注入。声明式与命令式两种用法，无 CGo 依赖。

使用方式

Goed UI 以声明式为主：应用界面通过 runtime.H(tag, props) 声明 VNode 树，runtime 负责挂载、属性补丁、Flex 布局与响应式更新闭环（Vue 风格）。命令式作为补充，用于画布内部的自绘场景与尚未提供声明式元素的复杂组件。

声明式优先（推荐）

根组件在 setup 中返回 VNode 树，状态经 reactive ref 驱动重渲染，无需手动管理失效与重绘：

Go
var App = runtime.DefineComponent("MyApp",
    func(ctx *runtime.SetupContext) runtime.RenderFunc {
        count := runtime.Ref(0)
        return func() []*runtime.VNode {
            return []*runtime.VNode{
                runtime.H("vstack", runtime.Props{"gap": 8},
                    runtime.H("titlebar", runtime.Props{"title": "My App"}),
                    runtime.H("text", runtime.Props{
                        "text": fmt.Sprintf("点击了 %d 次", count.Get()),
                    }),
                    runtime.H("button", runtime.Props{
                        "text":   "+1",
                        "onClick": func() { count.Set(count.Get() + 1) },
                    }),
                    // 声明式画布：onDraw 内部仍可命令式绘制
                    runtime.H("canvas", runtime.Props{
                        "flex":   1,
                        "onDraw": drawScene,
                        "onEvent": handleEvent,
                    }),
                ),
            }
        }
    })

命令式（补充手段）

命令式用于两类场景：画布内部的自绘内容（onDraw 回调内直接调用组件 Render），以及尚未提供声明式元素的复杂组件（Editor 套件、预览器等以 session 形式在 canvas 内编排）。

Go
// 编辑器 App 的架构：声明式壳（App 组件）+ 命令式芯（session.draw）
var App = runtime.DefineComponent("Editor", func(ctx *runtime.SetupContext) runtime.RenderFunc {
    return func() []*runtime.VNode{
        return []*runtime.VNode{runtime.H("canvas", runtime.Props{
            "onDraw":  session.draw,  // 命令式编排在这里发生
            "onEvent": session.event,
        })}
    }
})

func (s *session) draw(ctx core.DrawingContext, b core.Rect) {
    s.titlebar.Render(ctx, 0, 0, b.W)
    s.tree.Render(ctx, 0, 40, sidebarW, b.H-40-statusH)
    s.tabs.Render(ctx, sidebarW, 40, b.W-sidebarW)
    if s.docKind == KindImage && s.imageView.Ready() {
        s.imageView.Render(ctx, editorX, editorY, editorW, editorH)
    } else {
        s.editor.Render(ctx, editorX, editorY, editorW, editorH)
    }
    s.status.Render(ctx, 0, b.H-s.status.Height, b.W)
}

绘制上下文扩展能力

平台通过接口探测参与扩展能力，组件自动回退：

  • DrawingContext — 基础形状/文本（所有平台必须）
  • ImageDrawingContext — 位图绘制（不支持时 ImageView 回退文字提示）
  • AlphaDrawingContext — 半透明混合（不支持时 Dialog 回退不透明遮罩）
  • clipStack — 嵌套裁剪（Canvas Push/Pop 不破坏外层裁剪区）

一次事件-绘制往返

  • Win32 消息 — FramelessWindow wndProc 收到 WM_* 转为 core.Event
  • 事件分发 — app 层 DispatchEvent 递归派发到焦点组件
  • 状态更新 — 组件修改状态，reactive effect 触发重渲染
  • WM_PAINT — GDIContext 双缓冲绘制，组件按区域 Render
  • 上屏 — EndDraw BitBlt 双缓冲到窗口 DC

按钮 Button

命令式元素: button

标准文本按钮，渲染由主题驱动，悬停/按下状态内置。

使用

Go
// 命令式：创建按钮并注册点击回调
btn := components.NewButton("打开文件", 0, 0, 120, 34)
btn.OnClick = func() { openDialog() }

// 事件分发（宿主转发鼠标事件）
if btn.HandleMouseDown(x, y) { return true }
btn.HandleMouseMove(x, y)  // 悬停高亮

// 绘制
btn.Render(ctx)

// 声明式等价写法
runtime.H("button", runtime.Props{
    "text":   "打开文件",
    "onClick": openDialog,
})

API

成员类型说明
NewButton(text, x, y, w, h)*Button创建按钮，初始位置尺寸可被布局覆盖
OnClickfunc()点击回调
HandleMouseDown/Up(x, y)bool命中测试；Up 且在按钮内时触发 OnClick
HandleMouseMove(x, y)()更新悬停高亮
Render(ctx)()按当前状态绘制

对话框 Dialog

命令式模态

模态遮罩对话框：半透明压暗全屏，居中面板展示标题与消息。

使用

Go
// 确认对话框（危险操作）
dlg := components.NewDialog("", "")
dlg.ShowConfirm("删除", "将文件移入回收站？", func() {
    doDelete()
})

// 普通信息对话框
about := components.NewDialog("About", "Goed - v0.2")
about.Show()

// 事件分发：模态打开时优先拦截
if dlg.IsOpen() {
    if ev.Type == core.EventMouseDown {
        dlg.HandleMouseDown(ev.X, ev.Y, winW, winH)
        return true
    }
    if ev.Type == core.EventKeyDown {
        return dlg.HandleKeyDown(ev.Key)
    }
}

API

方法说明
NewDialog(title, message)创建（内容可后续改写）
ShowConfirm(title, msg, onOK)确认对话框；OK 时执行回调
Show() / Hide() / IsOpen()显隐控制
HandleMouseDown(x, y, w, h)面板内按钮 / 面板外取消
HandleKeyDown(key)Enter 确认、Esc 关闭
Render(ctx, x, y, w, h)全屏遮罩 + 面板；支持 alpha 上下文真混合

dlg := components.NewInputDialog()

    if err := os.Rename(oldPath, newPath); err == nil {
        refreshTree()
    }
}
dlg.Show("重命名", "位置: "+dir, filepath.Base(oldPath))

// 模态事件分发（Enter 提交非空值 / Esc 取消）
switch ev.Type {
case core.EventKeyChar:
    if ev.Char >= 0x20 { dlg.HandleChar(ev.Char) }
case core.EventKeyDown:
    dlg.HandleKeyDown(ev.Key)
case core.EventMouseDown:
    dlg.HandleMouseDown(ev.X, ev.Y, winW, winH)

上下文菜单 ContextMenu

命令式

右键菜单：宿主按命中条目动态构建，点击外部自动关闭。

使用

Go
// 按命中对象动态构建菜单项
tree.OnContextMenu = func(entry *components.FileTreeEntry, x, y int32) []components.MenuItem {
    if entry == nil {
        return []components.MenuItem{
            {Label: "刷新", Enabled: true, OnClick: refresh},
        }
    }
    return []components.MenuItem{
        {Label: "打开", Enabled: true, OnClick: func() { open(entry.Path) }},
        {Label: "重命名", Enabled: true, OnClick: func() { rename(entry.Path) }},
        {Separator: true},
        {Label: "删除", Enabled: true, OnClick: func() { del(entry.Path) }},
    }
}

// 事件分发：菜单打开时优先消费点击
if tree.ContextMenu.IsVisible {
    if tree.ContextMenu.HandleMouseDown(x, y) {
        return true  // 命中菜单项即触发回调
    }
}

画布 Canvas

双模式元素: canvas

自定义绘制与输入的保留模式组件：OnDraw 获得绘制权，OnEvent 接收输入，支持脏区局部重绘、嵌套裁剪、鼠标捕获与键盘焦点。

使用

Go
// 命令式：全屏画布场景（编辑器主画布即此模式）
cv := components.NewCanvas()
cv.SetDraw(func(ctx core.DrawingContext, b core.Rect) {
    ctx.DrawRect(0, 0, b.W, b.H, bg, true)
    drawScene(ctx)
})
cv.Focusable = true   // 键盘事件按焦点分发
cv.MouseCapture = true // 拖拽期间持续接收 move/up

// 声明式（editor/App.go 的用法）
runtime.H("canvas", runtime.Props{
    "flex":       1,
    "focusable":  true,
    "mouseCapture": true,
    "onDraw":     session.draw,
    "onEvent":    session.event,
})

// 脏区局部重绘（静态画布优化）
cv2 := components.NewCanvas().SetDirtyTracking(true)
cv2.InvalidateRect(core.Rect{X: 40, Y: 20, W: 30, H: 25}) // 仅重绘该区域

API

成员说明
NewCanvas(bounds...)创建；bounds 可省略由布局分配
SetDraw(fn)func(ctx, bounds) 安装绘制回调

查找替换 FindReplace

命令式回调解耦

悬浮查找替换面板：通过 getter/apply 回调接入任意文本宿主，可独立测试。实时高亮匹配、循环导航与全部替换。

使用

Go
// 三个回调：取全文 / 执行替换 / 跳转高亮
fr := components.NewFindReplace(
    func() string { return ed.GetText() },
    func(line, col, length int, rep string) { ed.ReplaceRange(line, col, length, rep) },
    func(line, col, length int) { ed.GotoPosition(line, col, length) },
)

// 同步匹配高亮到编辑器 + 请求重绘
fr.OnChange = func() {
    ed.FindMatches = fr.Matches
    ed.CurrentMatch = fr.Current
    invalidate()
}

// 打开（Ctrl+F）/ 打开替换（Ctrl+H）
fr.Show(ed.GetSelectedText())
fr.ToggleReplace()

// 绘制前锚定编辑器几何
fr.SetPlacement(editorX, editorY, editorW, editorH)
if fr.IsOpen() { fr.Render(ctx) }

编辑器左侧行号条：与滚动偏移联动，高亮光标行，折叠感知。

使用

Go
ln := components.NewLineNumbers()

// 每帧同步编辑器可见行窗口（editor/App.go 的 draw 路径）
rows := int(editorH / ed.LineHeight) + 1
ln.SetVisibleLines(rows)
ln.SetLineIndices(ed.VisibleLineWindow(rows))
ln.SetScroll(ed.ScrollLine + 1)
ln.ScrollOffset = ed.ScrollOffset
ln.Render(ctx, sidebarX, editorY, editorH)

API

方法说明

SetVisibleLines(rows) / SetLineIndices(idx)同步可见行数与实际行号窗口
SetScroll(scrollLine) / ScrollOffset滚动位置（整数行 + 小数偏移）
Render(ctx, x, y, h)渲染

标签页 Tabs

命令式声明式: 经 canvas 桥接

文档标签条：打开/关闭/激活，未保存圆点与 ⚠ 外部修改，拖拽重排，溢出滚动，右键批量关闭。

使用

Go
tabs := components.NewTabs()

// 宿主接线
tabs.OnSelect = func(path string) { openFile(path) }

tabs.OnNew = newDoc

tabs.Open(path)                          // 幂等：已打开则激活

tabs.SetExternalChange(path, true)       // ⚠ 外部修改

if active := tabs.ActivePath(); active != "" {
    openFile(active)
}

// 渲染与输入
tabs.Render(ctx, sidebarW, 40, winW-sidebarW)
tabs.HandleWheel(delta) // 横向滚动标签

图片预览 ImageView

命令式只读

PNG/JPEG/GIF 只读预览：魔数嗅探格式，等比缩放居中（最多 4 倍放大），Go 侧双线性缩放 + GDI 1:1 块传输，透明区棋盘格。

使用

Go
iv := components.NewImageView()

// 加载：进入即 Reset，失败不残留旧像素
if err := iv.Load(path); err != nil {
    status.SetLeft("无法解码图片: " + err.Error())
    openBinary(path) // 回退十六进制预览
    return
}

// 状态栏元信息
status.SetRight(fmt.Sprintf("%s | %d×%d | %s",
    iv.Format, iv.SrcW, iv.SrcH, humanSize(iv.SizeBytes)))

// 渲染（宿主在 docKind == KindImage 时调用）
if iv.Ready() {
    iv.Render(ctx, editorX, editorY, editorW, editorH)
}

// 关闭标签页时释放内存（上限 128MB 的大图）
if closedPath == previewPath {
    iv.Reset()
}
实现要点：缩放在 Go 侧双线性（预乘 alpha 域）完成后 1:1 BitBlt 上屏，规避真实窗口 DC 上 SetDIBitsToDevice 大位图（约 >1300×976）整块静默失败的驱动缺陷——这是"最大化后图片空白"问题的根因修复。

API

成员说明
NewImageView()创建
Load(path) error读取解码（上限 128MB）；进入即 Reset
Reset()释放像素与合成缓存，回到未加载态
Ready() / Path / Format / SrcW / SrcH / SizeBytes状态与元信息
Render(ctx, x, y, w, h)背景 + 缩放合成绘制；无位图能力平台回退文字

Application ui/app

构建器入口

应用 shell：挂载声明式根组件、配置窗口、启动事件循环。窗口创建与消息泵保持主线程亲和。

使用

Go
// 完整启动流程（editor/main.go）
func Run(filePath string) error {
    title := "Goed R1 - Text Editor"
    if filePath != "" {
        title = filepath.Base(filePath) + " - Goed R1"
    }
    return app.CreateApp().
        Mount(App).          // 声明式根组件
        Prop("path", filePath). // 传给 setup 的 props
        Title(title).
        Size(1200, 800).
        HideTitleBar(true).  // 卸载默认标题栏（自绘壳层）
        Run()
}

// 根组件定义（声明式）
var App = runtime.DefineComponent("Editor",
    func(ctx *runtime.SetupContext) runtime.RenderFunc {
        path := ctx.Props().Str("path", "")
        session := newSession(path)
        return func() []*runtime.VNode {
            return []*runtime.VNode{
                runtime.H("canvas", runtime.Props{
                    "flex": 1, "focusable": true,
                    "onDraw": session.draw, "onEvent": session.event,
                }),
            }
        }
    })

API

方法说明
CreateApp(root...)创建（默认 800×600）
Mount(def) / Prop(k, v)挂载根组件 / 传参
Title(s) / Size(w, h) / HideTitleBar(b) / WithOptions(opts)窗口配置
Run() error阻塞运行至窗口关闭
Run(title, w, h, def) / RunWithOptions(...)便捷函数
